小课表网站源码

技术说明：
这个网站不是单个静态 HTML 文件，而是一个 React + TypeScript（TSX）网站。
页面结构和交互写在 TSX 文件中，视觉样式写在 CSS 文件中，并由 Vite 构建。
浏览器最终运行时仍会生成标准的 HTML、CSS 和 JavaScript。

主要功能：
1. 周一至周日、每天九节课的课程表。
2. 点击课程格可填写课程名称、教室和自定义颜色。
3. 可新增、修改和删除课程。
4. 数据自动保存在当前浏览器的 localStorage 中。

以下为主要源文件（不包含自动生成的依赖锁文件和构建产物）。

'use client';

import { FormEvent, useEffect, useMemo, useState } from 'react';

const DAYS = [
  { short: 'MON', label: '周一' },
  { short: 'TUE', label: '周二' },
  { short: 'WED', label: '周三' },
  { short: 'THU', label: '周四' },
  { short: 'FRI', label: '周五' },
] as const;

const PERIODS = [
  { number: 1, time: '08:00–08:40', phase: '上午' },
  { number: 2, time: '09:10–09:50', phase: '上午' },
  { number: 3, time: '10:00–10:40', phase: '上午' },
  { number: 4, time: '10:50–11:30', phase: '上午' },
  { number: 5, time: '12:40–13:20', phase: '下午' },
  { number: 6, time: '13:30–14:10', phase: '下午' },
  { number: 7, time: '14:25–15:05', phase: '下午' },
  { number: 8, time: '15:15–15:55', phase: '下午' },
  { number: 9, time: '16:05–16:45', phase: '下午' },
] as const;

const COLORS = [
  { value: 'mint', label: '薄荷绿' },
  { value: 'sky', label: '晴空蓝' },
  { value: 'peach', label: '蜜桃橙' },
  { value: 'lavender', label: '薰衣草' },
  { value: 'lemon', label: '柠檬黄' },
  { value: 'rose', label: '珊瑚粉' },
] as const;

type Course = {
  name: string;
  room: string;
  color: (typeof COLORS)[number]['value'];
};

type Schedule = Record<string, Course>;
type Slot = { day: number; period: number };

const INITIAL_SCHEDULE: Schedule = {
  '0-0': { name: '高等数学', room: '明理楼 201', color: 'mint' },
  '2-1': { name: '大学英语', room: '博学楼 305', color: 'sky' },
  '4-2': { name: '计算机基础', room: '实验楼 A102', color: 'lavender' },
  '1-4': { name: '中国近代史', room: '致远楼 108', color: 'peach' },
  '3-6': { name: '体育', room: '东区操场', color: 'lemon' },
  '0-8': { name: '设计思维', room: '创意工坊 2', color: 'rose' },
};

const STORAGE_KEY = 'xiaokebiao-schedule-v1';

function slotKey(day: number, period: number) {
  return `${day}-${period}`;
}

export default function Home() {
  const [schedule, setSchedule] = useState<Schedule>(INITIAL_SCHEDULE);
  const [loaded, setLoaded] = useState(false);
  const [editing, setEditing] = useState<Slot | null>(null);
  const [courseName, setCourseName] = useState('');
  const [room, setRoom] = useState('');
  const [color, setColor] = useState<Course['color']>('mint');

  useEffect(() => {
    try {
      const saved = window.localStorage.getItem(STORAGE_KEY);
      if (saved) setSchedule(JSON.parse(saved) as Schedule);
    } catch {
      // If storage is unavailable, the timetable stays usable in memory.
    } finally {
      setLoaded(true);
    }
  }, []);

  useEffect(() => {
    if (!loaded) return;
    try {
      window.localStorage.setItem(STORAGE_KEY, JSON.stringify(schedule));
    } catch {
      // Edits still work for the current session if storage is unavailable.
    }
  }, [loaded, schedule]);

  useEffect(() => {
    if (!editing) return;
    const closeOnEscape = (event: KeyboardEvent) => {
      if (event.key === 'Escape') setEditing(null);
    };
    window.addEventListener('keydown', closeOnEscape);
    return () => window.removeEventListener('keydown', closeOnEscape);
  }, [editing]);

  const courseCount = useMemo(() => Object.keys(schedule).length, [schedule]);

  function openEditor(day: number, period: number) {
    const current = schedule[slotKey(day, period)];
    setEditing({ day, period });
    setCourseName(current?.name ?? '');
    setRoom(current?.room ?? '');
    setColor(current?.color ?? 'mint');
  }

  function openFirstAvailable() {
    for (let period = 0; period < PERIODS.length; period += 1) {
      for (let day = 0; day < DAYS.length; day += 1) {
        if (!schedule[slotKey(day, period)]) {
          openEditor(day, period);
          return;
        }
      }
    }
    openEditor(0, 0);
  }

  function saveCourse(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    if (!editing || !courseName.trim()) return;
    setSchedule((current) => ({
      ...current,
      [slotKey(editing.day, editing.period)]: {
        name: courseName.trim(),
        room: room.trim(),
        color,
      },
    }));
    setEditing(null);
  }

  function deleteCourse() {
    if (!editing) return;
    const key = slotKey(editing.day, editing.period);
    setSchedule((current) => {
      const next = { ...current };
      delete next[key];
      return next;
    });
    setEditing(null);
  }

  function clearSchedule() {
    if (courseCount === 0 || !window.confirm('确定要清空所有课程吗？')) return;
    setSchedule({});
  }

  const editingCourse = editing ? schedule[slotKey(editing.day, editing.period)] : undefined;

  return (
    <main className="site-shell">
      <div className="decor decor-one" aria-hidden="true" />
      <div className="decor decor-two" aria-hidden="true" />

      <header className="topbar">
        <a className="brand" href="#timetable" aria-label="小课表首页">
          <span className="brand-mark" aria-hidden="true">课</span>
          <span>
            <strong>小课表</strong>
            <small>把每一天安排得刚刚好</small>
          </span>
        </a>
        <div className="top-actions">
          <button className="button button-quiet" type="button" onClick={clearSchedule}>清空课表</button>
          <button className="button button-primary" type="button" onClick={openFirstAvailable}>
            <span aria-hidden="true">＋</span> 新增课程
          </button>
        </div>
      </header>

      <section className="intro" aria-labelledby="page-title">
        <div>
          <p className="eyebrow"><span /> 我的学习周</p>
          <h1 id="page-title">这一周，清晰可见</h1>
          <p className="intro-copy">点击任意格子，填入课程、教室，再选一个喜欢的颜色。</p>
        </div>
        <div className="week-summary" aria-label={`本周已有 ${courseCount} 节课程`}>
          <span className="summary-number">{courseCount}</span>
          <span>节课程<br /><small>已安排</small></span>
        </div>
      </section>

      <section className="schedule-card" id="timetable" aria-label="每周课程表">
        <div className="schedule-scroll">
          <div className="schedule-grid">
            <div className="corner-cell">
              <span>时间</span>
              <strong>星期</strong>
            </div>
            {DAYS.map((day, index) => (
              <div className="day-heading" key={day.short}>
                <span>{day.short}</span>
                <strong>{day.label}</strong>
                {index === 0 && <em>今天</em>}
              </div>
            ))}

            {PERIODS.map((period, periodIndex) => (
              <div className={`schedule-row row-${periodIndex + 1}`} key={period.number}>
                <div className="time-cell">
                  <span>第 {period.number} 节</span>
                  <strong>{period.time}</strong>
                  <small>{period.phase}</small>
                </div>
                {DAYS.map((day, dayIndex) => {
                  const course = schedule[slotKey(dayIndex, periodIndex)];
                  return (
                    <button
                      type="button"
                      className={`course-slot ${course ? `has-course course-${course.color}` : ''}`}
                      key={day.short}
                      onClick={() => openEditor(dayIndex, periodIndex)}
                      aria-label={course
                        ? `${day.label}第${period.number}节，${course.name}${course.room ? `，${course.room}` : ''}，点击编辑`
                        : `${day.label}第${period.number}节，点击添加课程`}
                    >
                      {course ? (
                        <span className="course-content">
                          <strong>{course.name}</strong>
                          <small>{course.room || '教室待定'}</small>
                        </span>
                      ) : (
                        <span className="add-hint" aria-hidden="true">＋</span>
                      )}
                    </button>
                  );
                })}
              </div>
            ))}
          </div>
        </div>
        <footer className="schedule-footer">
          <span><i className="status-dot" /> 已自动保存到当前浏览器</span>
          <span>轻点课程即可修改</span>
        </footer>
      </section>

      <p className="page-note">愿每一堂课，都离理想更近一点。</p>

      {editing && (
        <div className="modal-backdrop" role="presentation" onMouseDown={() => setEditing(null)}>
          <section
            className="course-modal"
            role="dialog"
            aria-modal="true"
            aria-labelledby="modal-title"
            onMouseDown={(event) => event.stopPropagation()}
          >
            <div className="modal-heading">
              <div>
                <p>{DAYS[editing.day].label} · 第 {PERIODS[editing.period].number} 节</p>
                <h2 id="modal-title">{editingCourse ? '编辑课程' : '添加课程'}</h2>
              </div>
              <button type="button" className="close-button" onClick={() => setEditing(null)} aria-label="关闭">×</button>
            </div>

            <form onSubmit={saveCourse}>
              <label className="field">
                <span>课程名称</span>
                <input
                  autoFocus
                  value={courseName}
                  onChange={(event) => setCourseName(event.target.value)}
                  placeholder="例如：高等数学"
                  maxLength={24}
                  required
                />
              </label>
              <label className="field">
                <span>上课教室</span>
                <input
                  value={room}
                  onChange={(event) => setRoom(event.target.value)}
                  placeholder="例如：明理楼 201"
                  maxLength={32}
                />
              </label>

              <fieldset className="color-field">
                <legend>课程颜色</legend>
                <div className="color-options">
                  {COLORS.map((item) => (
                    <label key={item.value} title={item.label}>
                      <input
                        type="radio"
                        name="course-color"
                        value={item.value}
                        checked={color === item.value}
                        onChange={() => setColor(item.value)}
                      />
                      <span className={`color-swatch swatch-${item.value}`}>
                        <i aria-hidden="true">✓</i>
                      </span>
                      <small>{item.label}</small>
                    </label>
                  ))}
                </div>
              </fieldset>

              <div className="modal-actions">
                {editingCourse && (
                  <button className="button button-danger" type="button" onClick={deleteCourse}>删除课程</button>
                )}
                <span className="action-spacer" />
                <button className="button button-quiet" type="button" onClick={() => setEditing(null)}>取消</button>
                <button className="button button-primary" type="submit">保存课程</button>
              </div>
            </form>
          </section>
        </div>
      )}
    </main>
  );
}

import { sites } from '@openai/sites-vite-plugin';
import tailwindcss from '@tailwindcss/postcss';
import vinext from 'vinext';
import { defineConfig } from 'vite';
import hostingConfig from './.openai/hosting.json';

const SITE_CREATOR_PLACEHOLDER_DATABASE_ID =
  '00000000-0000-4000-8000-000000000000';

const { d1, r2 } = hostingConfig;

// macOS Seatbelt blocks FSEvents, so Codex previews need polling for HMR.
const isCodexSeatbeltSandbox = process.env.CODEX_SANDBOX === 'seatbelt';

const localBindingConfig = {
  main: 'vinext/server/app-router-entry',
  compatibility_flags: ['nodejs_compat'],
  d1_databases: d1
    ? [
        {
          binding: d1,
          database_name: 'site-creator-d1',
          database_id: SITE_CREATOR_PLACEHOLDER_DATABASE_ID,
        },
      ]
    : [],
  r2_buckets: r2
    ? [
        {
          binding: r2,
          bucket_name: 'site-creator-r2',
        },
      ]
    : [],
};

export default defineConfig(async () => {
  // Keep Wrangler and Miniflare state project-local. These are non-secret tool
  // settings; application environment belongs in ignored `.env*` files.
  process.env.WRANGLER_WRITE_LOGS ??= 'false';
  process.env.WRANGLER_LOG_PATH ??= '.wrangler/logs';
  process.env.MINIFLARE_REGISTRY_PATH ??= '.wrangler/registry';

  // Wrangler snapshots its log path while the Cloudflare plugin is imported.
  const { cloudflare } = await import('@cloudflare/vite-plugin');

  return {
    css: { postcss: { plugins: [tailwindcss()] } },
    server: isCodexSeatbeltSandbox
      ? { watch: { useFsEvents: false, usePolling: true } }
      : undefined,
    plugins: [
      vinext(),
      sites(),
      cloudflare({
        viteEnvironment: { name: 'rsc', childEnvironments: ['ssr'] },
        config: localBindingConfig,
      }),
    ],
  };
});
